CREATE DATABASE SISTEMA;
USE SISTEMA;

CREATE TABLE GRUPO_USUARIO(
IDGRUPO INT PRIMARY KEY AUTO_INCREMENT NOT NULL,
NOME VARCHAR(50) UNIQUE NOT NULL,
DESCRICAO VARCHAR(100) NOT NULL,
CRIADO DATETIME NOT NULL,
ALTERADO DATETIME NOT NULL
);

CREATE TABLE USUARIOS(
IDUSUARIO INT PRIMARY KEY AUTO_INCREMENT NOT NULL,
NOME VARCHAR(100) NOT NULL,
EMAIL VARCHAR(100) UNIQUE NOT NULL,
SENHA VARCHAR(100) NOT NULL,
IDGRUPO INT NOT NULL,
CRIADO DATETIME NOT NULL,
FOREIGN KEY (IDGRUPO) REFERENCES GRUPO_USUARIO(IDGRUPO)
);

CREATE TABLE FORNECEDORES(
ID INT PRIMARY KEY AUTO_INCREMENT NOT NULL,
NOME VARCHAR(100) NOT NULL,
ESTADO VARCHAR(4) NOT NULL,
CIDADE VARCHAR(50) NOT NULL,
BAIRRO VARCHAR(50) NOT NULL,
NUMERO VARCHAR(20) NOT NULL,
CNPJ VARCHAR(20) NOT NULL,
);


CREATE TABLE PRODUTOS_VENDA(
IDPRODUTO INT PRIMARY KEY AUTO_INCREMENT NOT NULL,
NOME VARCHAR(60) UNIQUE NOT NULL,
DESCRICAO VARCHAR(100),
PRECO DECIMAL(8,3) NOT NULL,
IMAGEM VARCHAR(50),
EAN VARCHAR(20),
CRIADO DATETIME NOT NULL,
IDFORNECEDOR INT NOT NULL
);
